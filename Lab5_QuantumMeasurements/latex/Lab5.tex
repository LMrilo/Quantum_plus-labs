\section{Introduction and Objectives}

This laboratory offers an \textbf{experimental verification and visualization} of the
fundamental quantum concepts presented in the \textit{Fundamental Concepts of Quantum
Technologies} course. You will explore \textbf{superposition}, \textbf{quantum
measurement}, \textbf{photon polarization} and \textbf{electron spin}, all represented on
the Bloch sphere.

Through guided simulations you will analyze how measurement affects quantum states, how
probabilistic outcomes emerge, and how spin and polarization can encode and carry quantum
information. The interactive Bloch sphere on the lab's page lets you build intuition for
the geometry of quantum states and for the meaning of measurement in different bases.

By the end of the lab you will be able to connect abstract quantum principles to
measurable phenomena, describe how measurement defines information in quantum systems,
and interpret visual representations of qubit states in terms of real physical
quantities.

\section{Theoretical Background}

\subsection{The Measurement Postulate (Born Rule)}

For a qubit in state $\ket{\psi}$ measured in the basis $\{\ket{m}\}$, the probability
of outcome $m$ follows the Born rule:
\begin{equation}
    P(m \mid \psi) = |\braket{m}{\psi}|^2 .
    \label{eq:born}
\end{equation}
A \emph{basis} is the pair of perfectly distinguishable states a measurement question can
answer, such as $\{\ket{0}, \ket{1}\}$. Here $\braket{m}{\psi}$ is the \emph{amplitude}
of $\ket{m}$ in $\ket{\psi}$. If $\ket{\psi} = a\ket{0} + b\ket{1}$ and
$\ket{m} = c\ket{0} + d\ket{1}$, then $\braket{m}{\psi} = c^{*}a + d^{*}b$, where $c^{*}$
is the complex conjugate of $c$ (change $i$ into $-i$). Example: for
$\ket{m} = \ket{+} = (\ket{0} + \ket{1})/\sqrt 2$,
$\braket{+}{\psi} = (a + b)/\sqrt 2$. The post-measurement state depends on which
procedure is used to perform the measurement (Section~\ref{sec:procedures}).

\subsection{Bloch Sphere Parametrization}

Any single-qubit pure state can be written as
\begin{equation}
    \ket{\psi} = \cos\!\left(\frac{\theta}{2}\right)\ket{0}
    + e^{i\varphi}\sin\!\left(\frac{\theta}{2}\right)\ket{1},
    \qquad \theta \in [0,\pi],\quad \varphi \in [0, 2\pi).
    \label{eq:bloch}
\end{equation}
The two amplitudes are $\cos(\theta/2)$ and $e^{i\varphi}\sin(\theta/2)$, so a
measurement in the $Z$-basis gives $P(0) = \cos^2(\theta/2)$ and
$P(1) = \sin^2(\theta/2)$: the polar angle $\theta$ alone fixes them. The factor
$e^{i\varphi}$ is a \emph{relative phase}: invisible to a $Z$-measurement, visible in the
$X$- and $Y$-bases (Activity 2). The half angle $\theta/2$ is what makes opposite points of
the sphere ($\theta = 0$ and $\theta = \pi$) perfectly distinguishable states. (The theory
pill QC03 --- \textit{The qubit and the Bloch sphere} introduces this picture.)

\begin{table}[H]
\centering
\begin{tabular}{lcccc}
\toprule
state & written out & $\theta$ & $\varphi$ & position \\
\midrule
$\ket{0}$ & $\ket{0}$ & $0$ & --- & North pole \\
$\ket{1}$ & $\ket{1}$ & $\pi$ & --- & South pole \\
$\ket{+}$ & $(\ket{0} + \ket{1})/\sqrt 2$ & $\pi/2$ & $0$ & $+X$ \\
$\ket{-}$ & $(\ket{0} - \ket{1})/\sqrt 2$ & $\pi/2$ & $\pi$ & $-X$ \\
$\ket{+i}$ & $(\ket{0} + i\ket{1})/\sqrt 2$ & $\pi/2$ & $\pi/2$ & $+Y$ \\
$\ket{-i}$ & $(\ket{0} - i\ket{1})/\sqrt 2$ & $\pi/2$ & $3\pi/2$ & $-Y$ \\
\bottomrule
\end{tabular}
\caption{The six cardinal states of the Bloch sphere.}
\label{tab:cardinal}
\end{table}

\subsection{Pauli Operators and Eigenstates}

\begin{equation}
    X = \begin{pmatrix}0&1\\1&0\end{pmatrix},\qquad
    Y = \begin{pmatrix}0&-i\\i&0\end{pmatrix},\qquad
    Z = \begin{pmatrix}1&0\\0&-1\end{pmatrix}.
\end{equation}
Measuring in the $Z$-basis gives the outcomes $\ket{0}$ or $\ket{1}$; in the $X$-basis,
$\ket{+}$ or $\ket{-}$; in the $Y$-basis, $\ket{+i}$ or $\ket{-i}$. ($X$ and $Z$ are the
gates of the theory pill QC04 --- \textit{Single-qubit gates}; here they label the three
measurement questions.) The states that give a \emph{certain} answer to a question are
called its \emph{eigenstates}. No state gives a certain answer to both the $Z$ and the
$X$ question: they are \textbf{incompatible}, so sequential measurements in these bases
destroy prior information. (Optional, for readers who know matrix products:
$[X, Z] = XZ - ZX = -2iY \neq 0$.)

\subsection{Two Measurement Procedures}
\label{sec:procedures}

\begin{procedurebox}
\textbf{Same probabilities, different post-states.} Throughout this lab, distinguish
carefully between two operationally different ways of measuring in a non-$Z$ basis $B$:
\begin{itemize}
    \item \textbf{Procedure 1 --- direct projection.} Project $\ket{\psi}$ directly onto
          the eigenstates of $B$. For an outcome $m \in B$, the post-measurement state is
          $\ket{m}$, an eigenstate of $B$.
    \item \textbf{Procedure 2 --- rotate, then $Z$-measure.} Apply a reversible gate (a
          \emph{unitary}) $U_B$ that turns the two states of $B$ into $\ket{0}$ and
          $\ket{1}$ (the \emph{computational} basis), then perform a $Z$-measurement.
          Outcomes are 0 or 1, and the post-measurement state is $\ket{0}$ or $\ket{1}$
          --- a $Z$-eigenstate, \emph{not} a $B$-eigenstate.
\end{itemize}
The canonical rotations are
\[
    U_X = H, \qquad U_Y = H \cdot S^\dagger, \qquad U_Z = I .
\]
Here $H$ is the Hadamard gate of the theory pill QC04 --- \textit{Single-qubit gates}
($H\ket{+} = \ket{0}$, $H\ket{-} = \ket{1}$); $S^\dagger$ multiplies the amplitude of
$\ket{1}$ by $-i$, which turns $\ket{\pm i}$ into $\ket{\pm}$; $I$ does nothing.

\medskip
The two procedures give the \textbf{same outcome probabilities} (Born rule), but the
post-measurement states differ. Gate-based quantum processors (such as IBM's) can only
execute Procedure 2. Some physical devices measure directly in their own basis and so
realize Procedure 1: a polarizer (Activity 3), a Stern--Gerlach magnet (Activity 4), and
the simulator on the lab's page.
\end{procedurebox}

\section{Activities}

\subsection{Activity 1: State Preparation \& Bloch Sphere}

\begin{studentbox}
\textbf{Goal.} Build visual intuition for qubit states by placing them on the Bloch
sphere and understanding the parametrization $(\theta, \varphi)$ of
equation~\eqref{eq:bloch} (the six cardinal states are in Table~\ref{tab:cardinal}).

\medskip
\textbf{Instructions} (the simulator on the lab's page):
\begin{enumerate}
    \item Click each quick-state button and observe where the vector lands.
    \item Drag the $\theta$ slider from $0^\circ$ to $180^\circ$ and watch the vector travel
          from the north pole to the south pole.
    \item Fix $\theta = 90^\circ$ and drag the $\varphi$ slider from $0^\circ$ to $360^\circ$.
          The vector traces the equator, passing through $\ket{+}$, $\ket{+i}$, $\ket{-}$,
          $\ket{-i}$ and back.
    \item Note that changing $\varphi$ alone does not change the $Z$-basis probabilities.
          $\varphi$ is a \textbf{relative phase}: it changes the state ($\varphi = \pi$
          turns $\ket{+}$ into $\ket{-}$) but is invisible to a measurement in the
          $Z$-basis --- Activity 2 makes it visible in the $X$-basis.
\end{enumerate}

\textbf{Reflection.} Why do $\ket{+}$ and $\ket{-}$ both have $\theta = \pi/2$ but lie on
opposite sides of the equator? What does $\varphi$ physically represent?
\end{studentbox}

\keepspace{8\baselineskip}
\subsection{Activity 2: Measurement in Different Bases}

\begin{studentbox}
\textbf{Goal.} Measure the same state in the $Z$, $X$ and $Y$ bases and observe how the
measurement basis determines the outcome probabilities.

\medskip
\textbf{Theory --- the Born rule in practice.} Prepare
$\ket{\psi} = \ket{+} = (\ket{0} + \ket{1})/\sqrt 2$. Measuring in the $Z$-basis:
$P(0) = P(1) = 50\%$. Measuring in the $X$-basis: $P(+) = 100\%$, $P(-) = 0\%$, because
$\braket{+}{+} = 1$ and $\braket{-}{+} = 0$.

\medskip
\textbf{Instructions} (the simulator on the lab's page):
\begin{enumerate}
    \item Click $\ket{+}$. The probability bars show 50\% / 50\% in the $Z$-basis ---
          verify this.
    \item Switch to the \textbf{$X$-basis}. The bars now show 100\% / 0\%. Click
          \textbf{Measure!} several times --- always the same outcome. Why?
    \item Switch to the \textbf{$Y$-basis}. The bars return to 50\% / 50\%. Click
          Measure! once and note the outcome. Click it again: the same outcome repeats,
          because the first measurement already left the qubit in that $Y$ state. To see
          the 50/50 randomness, click $\ket{+}$ again before \emph{every} measurement
          (this is called re-preparing the state).
    \item Now try $\ket{-}$: same $\theta$, different $\varphi$. In the $Z$-basis: still
          50/50. In the $X$-basis: the \emph{other} eigenstate becomes certain. In the
          $Y$-basis: 50/50 again.
    \item For each trial, record: state $\to$ basis $\to$ probability bars $\to$ observed
          outcome. Watch how the vector snaps to the collapsed position after each
          measurement.
\end{enumerate}

\textbf{Reflection.} Why does measuring $\ket{+}$ in the $X$-basis always give the same
result? How does this change if you prepare $\ket{-}$ instead?
\end{studentbox}

\keepspace{8\baselineskip}
\subsection{Activity 3: Photon Polarization}

\begin{studentbox}
\textbf{Goal.} Map photon polarization to qubit states, and observe how a polarizer acts
as a measurement device and causes state collapse.

\medskip
\textbf{Theory --- polarization as a qubit.} Horizontal polarization $\leftrightarrow
\ket{0}$, vertical polarization $\leftrightarrow \ket{1}$, diagonal ($45^\circ$)
polarization $\leftrightarrow \ket{+} = (\ket{0} + \ket{1})/\sqrt 2$. Linear polarization at
angle $\alpha$ (from the horizontal) is
$\ket{\alpha} = \cos\alpha\,\ket{0} + \sin\alpha\,\ket{1}$. On the Bloch sphere it sits at
$\theta = 2\alpha$, $\varphi = 0$ for $0^\circ \le \alpha \le 90^\circ$ (for $90^\circ < \alpha < 180^\circ$ at
$\theta = 360^\circ - 2\alpha$, $\varphi = 180^\circ$: the anti-diagonal $135^\circ$ is $\ket{-}$ up to a
sign, at $-X$): turning the polarization by $90^\circ$ (horizontal to
vertical) moves the point by $180^\circ$ (north to south pole).

A polarizer oriented at angle $\alpha$ measures in the basis
$\{\ket{\alpha}, \ket{\alpha + 90^\circ}\}$ by \emph{Procedure 1}: a photon that passes
is left polarized at $\alpha$; one that does not pass is absorbed. A photon with initial
polarization angle $\alpha_0$ passes with probability
\begin{equation}
    P(\text{pass}) = \cos^2(\alpha - \alpha_0)
\end{equation}
--- this is Malus's law. For a beam of light it gives the transmitted intensity; for a
single photon it is the Born rule: each photon either passes or is absorbed.

\medskip
\textbf{Instructions} (Quantum Flytrap Virtual Lab, \url{https://lab.quantumflytrap.com/}):
\begin{enumerate}
    \item Open Quantum Flytrap and select a single-photon polarization experiment.
    \item Prepare a \textbf{horizontally polarized} photon and pass it through a
          \textbf{vertical} polarizer. Note: 0\% transmission.
    \item Prepare a \textbf{diagonally polarized} photon ($45^\circ$) and measure it with a
          vertical polarizer. Note: 50\% transmission.
    \item After passing a $45^\circ$ photon through a vertical polarizer (which leaves it in
          $\ket{1}$ when it passes), measure again with a horizontal polarizer. Note: 0\%
          transmission. The state has collapsed to $\ket{1}$, which is orthogonal to
          horizontal.
    \item Record each measurement and explain what state the photon is in before and after
          each polarizer.
\end{enumerate}

\textbf{Reflection.} Why does inserting a polarizer at $45^\circ$ between two crossed
polarizers (horizontal, then vertical) let some light through, even though the crossed
polarizers alone block 100\% of the light? (Hint: state collapse resets the photon's
information.)
\end{studentbox}

\keepspace{8\baselineskip}
\subsection{Activity 4: Electron Spin \& Stern--Gerlach}

\begin{studentbox}
\textbf{Goal.} Observe spin quantization and understand how spin-up and spin-down states
along different axes map to qubit states and Bloch sphere positions.

\medskip
\textbf{Theory --- spin and the Bloch sphere.} An electron's spin is a two-level quantum
system --- exactly like a qubit. (Stern and Gerlach's 1922 experiment used silver atoms,
whose spin is that of their single outer electron; that is why the instructions below speak
of atoms.) Spin-up along $Z$ corresponds to $\ket{0}$ (north pole),
spin-down to $\ket{1}$ (south pole). Spin-up along $X$ corresponds to $\ket{+}$ (equator,
$+X$); spin-up along $Y$ to $\ket{+i}$ ($+Y$). A Stern--Gerlach apparatus oriented along
an axis splits a beam into two outputs (spin-up and spin-down for that axis), each with
the probability given by the Born rule~\eqref{eq:born}. Each apparatus implements
\emph{Procedure 1}: the post-measurement state is the spin state of the chosen axis that
was found.

\medskip
\textbf{Instructions} (the simulator on the lab's page). A spin behaves exactly like the
qubit on the Bloch simulator: a $Z$-oriented apparatus is a measurement in the $Z$-basis,
an $X$-oriented one a measurement in the $X$-basis, and clicking $\ket{0}$ sends a fresh
atom from the spin-up output of a first $Z$ apparatus. If your browser can still run it,
the PhET Stern--Gerlach simulation (\url{https://phet.colorado.edu/en/simulations/stern-gerlach};
a legacy Flash simulation) shows the same experiment with beams.
\begin{enumerate}
    \item Click $\ket{0}$ and measure in the \textbf{$Z$-basis}, several times (click
          $\ket{0}$ before each): always spin-up. A second $Z$ apparatus confirms the first.
    \item Click $\ket{0}$ and measure in the \textbf{$X$-basis} instead, ten times (click
          $\ket{0}$ before each): about half $\ket{+}$, half $\ket{-}$. Why? Because
          $\ket{0} = (\ket{+} + \ket{-})/\sqrt 2$ (check: $\ket{+} + \ket{-} = \tfrac{2}{\sqrt 2}\,\ket{0}
          = \sqrt 2\,\ket{0}$): each of the two terms has amplitude $1/\sqrt 2$, so each outcome
          has probability $\tfrac12$.
    \item Now take an atom that came out $\ket{+}$ and measure it in the \textbf{$Z$-basis}
          (a third apparatus). Repeat the whole sequence a few times (click $\ket{0}$,
          measure in $X$ until the outcome is $\ket{+}$, then measure in $Z$): $\ket{0}$ and
          $\ket{1}$ come out about 50/50 --- the $X$ measurement erased the $Z$ information.
\end{enumerate}

\textbf{Reflection.} How does this sequential Stern--Gerlach experiment show that the
spin-$z$ and spin-$x$ questions are incompatible? What would you expect if a spin could
have a definite $z$ value and a definite $x$ value at the same time?
\end{studentbox}

\keepspace{8\baselineskip}
\subsection{Activity 5: Sequential Measurements \& State Collapse}

\begin{studentbox}
\textbf{Goal.} Demonstrate that measuring in $Z$, then $X$, then $Z$ again produces random
outcomes in the final $Z$ measurement, even when the first $Z$ gave a definite result.

\medskip
\textbf{Theory --- incompatible questions.} The $Z$ question (``$\ket{0}$ or $\ket{1}$?'')
and the $X$ question (``$\ket{+}$ or $\ket{-}$?'') have no common eigenstate:
$\ket{0} = (\ket{+} + \ket{-})/\sqrt 2$ and $\ket{+} = (\ket{0} + \ket{1})/\sqrt 2$, so a
state that answers one question for certain answers the other with 50/50. (Optional, for
readers who know matrix products:
$XZ - ZX = \begin{pmatrix}0&-2\\2&0\end{pmatrix} = -2iY \neq 0$.)

\medskip
\textbf{Which procedure?} The simulator on the lab's page measures by
\textbf{Procedure 1}: after the $X$-measurement the qubit is left in $\ket{+}$ or $\ket{-}$.
A gate-based processor would measure $X$ by \textbf{Procedure 2} --- apply $U_X = H$, then
$Z$-measure --- and leave $\ket{0}$ or $\ket{1}$ instead. The final $Z$ statistics are the
same either way (50/50 across many runs); what differs is the state in between.
Self-Assessment question 3 asks about the Procedure 2 version.

\medskip
\textbf{Instructions} (the simulator on the lab's page):
\begin{enumerate}
    \item Click $\ket{0}$, select the $Z$-basis and click \textbf{Measure!}: always $\ket{0}$.
          Post-state: $\ket{0}$.
    \item \textbf{$X$-measurement:} switch to the $X$-basis (the bars show 50\% / 50\%) and
          click Measure!. The vector jumps to $\ket{+}$ or $\ket{-}$ on the equator.
    \item \textbf{Second $Z$-measurement:} switch back to the $Z$-basis (the bars show
          50\% / 50\% again) and click Measure!: $\ket{0}$ or $\ket{1}$. The
          $X$-measurement erased the $Z$ information.
    \item Repeat steps 1--3 ten times (click $\ket{0}$ each time) and count how often the
          last outcome is $\ket{0}$: about half the time.
    \item \textbf{Compare to $Z \to Z \to Z$:} click $\ket{0}$ and measure in the $Z$-basis
          three times in a row. The outcome is always $\ket{0}$: a repeated question does
          not disturb the answer.
\end{enumerate}

\textbf{Reflection.} The probabilities of the final $Z$ measurement are 50/50 (across many
runs) because the $X$ step left the qubit in $\ket{+}$ or $\ket{-}$, each of which gives
0 or 1 with probability 1/2. Can the final $Z$ measurement ever recover the original $Z$
information? Why not? What property of the two questions makes this fundamental?
\end{studentbox}

\section{Self-Assessment}

\begin{reflectionbox}
\textbf{Question 1 --- the Born rule.} A qubit is prepared in the state
$\ket{\psi} = \frac{\sqrt 3}{2}\ket{0} + \frac12\ket{1}$. What is the probability of
measuring the outcome $\ket{1}$ in the $Z$-basis?
\begin{itemize}
    \item[(a)] $\frac{\sqrt 3}{2}$ \qquad (b) $\frac14$ \qquad (c) $\frac12$
\end{itemize}

\textbf{Question 2 --- state collapse: procedure matters.} Starting from
$\ket{\psi} = \ket{0}$, you apply $H$ and then perform a $Z$-basis measurement. The
outcome is 0. What is the post-measurement state?
\begin{itemize}
    \item[(a)] $\ket{+} = (\ket{0} + \ket{1})/\sqrt 2$
    \item[(b)] $\ket{0}$
    \item[(c)] $\frac12\ket{0} + \frac12\ket{1}$ (renormalized)
\end{itemize}

\textbf{Question 3 --- non-commutativity.} You prepare $\ket{0}$, measure $Z$ (get 0,
post-state $\ket{0}$), then measure $X$ via Procedure 2 (apply $H$, then measure in the
$Z$-basis), then measure $Z$ a final time. What are the probabilities for the final $Z$
measurement, over many runs (the middle outcome is random)?
\begin{itemize}
    \item[(a)] 100\% outcome 0 --- the initial $Z$ measurement fixed the state.
    \item[(b)] 50\% / 50\% --- the $X$ measurement randomized the qubit between $\ket{0}$
               and $\ket{1}$.
    \item[(c)] 0\% outcome 0 --- measuring $X$ always flips the $Z$ information.
\end{itemize}
\end{reflectionbox}
